# The species class the Viltrumite portraits hang off. A portrait set, a
# portrait category and a species-name list all key off a class, so this is
# what 00_viltrumite_portraits.txt, 00_viltrumite_portrait_categories.txt and
# 00_viltrumite_species_names.txt attach to.
#
# The prescripted Viltrum Empire is of this class, which is what makes the
# Viltrumite portraits its default rather than something to pick manually.
#
# Machine empires are excluded the same way vanilla excludes them from HUM:
# the portraits are organic bodies and read wrong on a machine.
#
# Needs VILTRUMITE, VILTRUMITE_desc and VILTRUMITE_plural in localisation.

VILTRUMITE = {
	archetype = BIOLOGICAL

	possible = {
		authority = {
			NOT = {
				value = auth_machine_intelligence
				text = SPECIES_CLASS_MUST_NOT_USE_MACHINE_INTELLIGENCE
			}
		}
	}

	possible_secondary = { always = yes }

	# Every biological class in vanilla carries this; without it a Viltrumite
	# species is not treated as organic by ascension paths and trait gates.
	trait = "trait_organic"

	graphical_culture = humanoid_01

	move_pop_sound_effect = "moving_pop_confirmation"

	gender = yes
}
